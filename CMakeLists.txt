cmake_minimum_required(VERSION 3.25)
project(StarlightNative LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "The current native renderer requires Windows / DirectX 12")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/utf-8 /EHsc)
add_library(starlight_editor_core STATIC editor/cpp/scene.cpp editor/cpp/scene_json.cpp editor/cpp/platform.cpp editor/cpp/mesh.cpp editor/cpp/geometry.cpp editor/cpp/assets.cpp editor/cpp/commands.cpp editor/cpp/flow.cpp editor/cpp/render_scene.cpp editor/cpp/projects.cpp editor/cpp/interactions.cpp editor/cpp/command_runner.cpp)
target_include_directories(starlight_editor_core PUBLIC editor/cpp)
target_link_libraries(starlight_editor_core PUBLIC ole32 shell32 bcrypt)
add_executable(Starlight.NativeCoreChecks editor/cpp/core_checks.cpp)
target_link_libraries(Starlight.NativeCoreChecks PRIVATE starlight_editor_core)
add_executable(Starlight.NativeCommands editor/cpp/cli.cpp)
target_link_libraries(Starlight.NativeCommands PRIVATE starlight_editor_core)
add_executable(Starlight.NativeDiagnosticChecks editor/cpp/diagnostic_checks.cpp)
target_link_libraries(Starlight.NativeDiagnosticChecks PRIVATE starlight_editor_core dbghelp)
add_executable(Starlight WIN32 deep/src/runtime.cpp)
set_target_properties(Starlight PROPERTIES CXX_STANDARD 17)
target_link_libraries(Starlight PRIVATE user32 shell32 comdlg32 dbghelp)
enable_testing()
add_test(NAME native-core COMMAND Starlight.NativeCoreChecks "${CMAKE_BINARY_DIR}/checks" "${CMAKE_SOURCE_DIR}/migration/geometry-reference.json")
add_test(NAME native-diagnostics COMMAND Starlight.NativeDiagnosticChecks "${CMAKE_BINARY_DIR}/diagnostic-checks/$<CONFIG>")
option(STARLIGHT_BUILD_RENDERER "Build the native D3D12 viewport and Jolt physics" ON)
if(STARLIGHT_BUILD_RENDERER)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME native-no-managed COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/migration/verify_no_managed.py")
 find_program(STARLIGHT_DXC dxc HINTS "C:/Program Files (x86)/Windows Kits/10/bin/10.0.26100.0/x64" REQUIRED)
 set(STARLIGHT_NGX_SDK "${CMAKE_SOURCE_DIR}/<separate-game-project>/starlight-realms/native-dx12/third_party/sdk/streamline/external/ngx-sdk" CACHE PATH "Native NGX SDK for 1:1 ray reconstruction")
 if(NOT EXISTS "${STARLIGHT_NGX_SDK}/include/nvsdk_ngx.h")
  message(FATAL_ERROR "Set STARLIGHT_NGX_SDK to the installed native NGX SDK")
 endif()
 set(shader_dir "${CMAKE_BINARY_DIR}/generated")
 add_custom_command(OUTPUT "${shader_dir}/render_device_editor.h" "${shader_dir}/atmosphere_shader.h" "${shader_dir}/dxr_shader.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/editor/native/cmake_shaders.py" "${shader_dir}" "${STARLIGHT_DXC}"
  DEPENDS editor/native/cmake_shaders.py editor/native/prepare_renderer.py editor/native/world_surface.hlsl deep/engine/render_device.h deep/engine/dxr.hlsl deep/engine/svgf.hlsl deep/engine/atmosphere.hlsl
  VERBATIM)
 add_library(Starlight.Viewport SHARED editor/native/viewport.cpp "${shader_dir}/render_device_editor.h" "${shader_dir}/atmosphere_shader.h" "${shader_dir}/dxr_shader.h"
  deep/engine/latency/antilag2.cpp deep/engine/model/skinned.cpp deep/engine/model/texture.cpp deep/engine/upscale/upscale_off.cpp deep/engine/upscale/upscale_stubs.cpp)
 target_compile_definitions(Starlight.Viewport PRIVATE STARLIGHT_CMAKE_RENDERER)
 target_include_directories(Starlight.Viewport PRIVATE "${shader_dir}" deep/engine "${STARLIGHT_NGX_SDK}/include")
 target_link_libraries(Starlight.Viewport PRIVATE user32 d3d12 dxgi d3dcompiler ws2_32 windowscodecs ole32 shell32 advapi32 wintrust crypt32 shlwapi
  debug "${STARLIGHT_NGX_SDK}/lib/Windows_x86_64/nvsdk_ngx_d_dbg.lib"
  optimized "${STARLIGHT_NGX_SDK}/lib/Windows_x86_64/nvsdk_ngx_d.lib")
 add_executable(Starlight.NativeRenderChecks WIN32 editor/native/standalone.cpp)
 add_executable(Starlight.NativeTextureChecks editor/cpp/texture_checks.cpp deep/engine/model/texture.cpp)
 target_link_libraries(Starlight.NativeTextureChecks PRIVATE d3d12 dxgi windowscodecs ole32)
 add_test(NAME native-textures COMMAND Starlight.NativeTextureChecks "${CMAKE_BINARY_DIR}/texture-checks/$<CONFIG>")
 target_link_libraries(Starlight.NativeRenderChecks PRIVATE Starlight.Viewport user32 shell32)
 add_subdirectory(deep/Physics physics)
 add_library(starlight_editor_session STATIC editor/cpp/physics.cpp editor/cpp/native_session.cpp)
 target_link_libraries(starlight_editor_session PUBLIC starlight_editor_core Starlight.Viewport Starlight.Physics)
 set_target_properties(Starlight.Physics PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
 add_executable(Starlight.NativeEditor WIN32 editor/cpp/editor_main.cpp)
 target_link_libraries(Starlight.NativeEditor PRIVATE starlight_editor_session comctl32 dbghelp winmm)
 add_test(NAME native-classic-ui COMMAND Starlight.NativeEditor --ui-check "${CMAKE_BINARY_DIR}/classic-ui-$<CONFIG>.json")
 add_executable(Starlight.NativeSessionChecks editor/cpp/session_checks.cpp)
 target_link_libraries(Starlight.NativeSessionChecks PRIVATE starlight_editor_session psapi)
 add_test(NAME native-session COMMAND Starlight.NativeSessionChecks "${CMAKE_BINARY_DIR}/session-checks/$<CONFIG>" "${CMAKE_SOURCE_DIR}")
 option(STARLIGHT_BUILD_PROJECTS "Build native Empty and game project modules" ON)
 if(STARLIGHT_BUILD_PROJECTS)
  foreach(project_name IN ITEMS Empty GameProject)
   add_library(${project_name} SHARED "deep/projects/${project_name}/project.cpp"
    deep/engine/latency/antilag2.cpp deep/engine/model/skinned.cpp deep/engine/model/texture.cpp
    deep/engine/upscale/upscale_off.cpp deep/engine/upscale/upscale_stubs.cpp)
   set_target_properties(${project_name} PROPERTIES CXX_STANDARD 17 RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/projects/${project_name}")
   target_include_directories(${project_name} PRIVATE deep/engine "deep/projects/${project_name}")
   target_link_libraries(${project_name} PRIVATE user32 d3d12 dxgi d3dcompiler ws2_32 windowscodecs ole32 shell32 winhttp)
   add_custom_command(TARGET ${project_name} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${CMAKE_SOURCE_DIR}/deep/projects/${project_name}/${project_name}.star" "$<TARGET_FILE_DIR:${project_name}>/${project_name}.star")
  endforeach()
  target_sources(GameProject PRIVATE deep/projects/GameProject/net/ws_client.cpp)
 endif()
endif()
