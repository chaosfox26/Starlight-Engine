cmake_minimum_required(VERSION 3.22)
project(StarlightPhysics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(TARGET_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(TARGET_HELLO_WORLD OFF CACHE BOOL "" FORCE)
set(TARGET_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
set(TARGET_SAMPLES OFF CACHE BOOL "" FORCE)
set(TARGET_VIEWER OFF CACHE BOOL "" FORCE)
set(STARLIGHT_CPU_BASELINE "portable" CACHE STRING "CPU instruction baseline: portable or avx2")
set_property(CACHE STARLIGHT_CPU_BASELINE PROPERTY STRINGS portable avx2)
if(NOT STARLIGHT_CPU_BASELINE STREQUAL "portable" AND NOT STARLIGHT_CPU_BASELINE STREQUAL "avx2")
 message(FATAL_ERROR "STARLIGHT_CPU_BASELINE must be portable or avx2")
endif()
set(starlight_extended_cpu OFF)
if(STARLIGHT_CPU_BASELINE STREQUAL "avx2")
 set(starlight_extended_cpu ON)
endif()
foreach(instruction SSE4_1 SSE4_2 AVX AVX2 LZCNT TZCNT F16C FMADD)
 set(USE_${instruction} ${starlight_extended_cpu} CACHE BOOL "Selected by Starlight CPU baseline" FORCE)
endforeach()
set(USE_AVX512 OFF CACHE BOOL "Not required by the Starlight CPU baseline" FORCE)
if(STARLIGHT_CPU_BASELINE STREQUAL "avx2" AND MSVC AND NOT CMAKE_VS_PLATFORM_NAME AND CMAKE_SIZEOF_VOID_P EQUAL 8)
 set(CMAKE_VS_PLATFORM_NAME x64)
endif()
add_subdirectory(../third_party/JoltPhysics/Build jolt)
if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
 # Jolt's headers rely on a transitive type_traits include with MSVC.
 target_compile_options(Jolt PRIVATE -include type_traits)
endif()
add_library(Starlight.Physics SHARED PhysicsWorld.cpp)
set_target_properties(Starlight.Physics PROPERTIES CXX_STANDARD ${STARLIGHT_CXX_STANDARD})
target_link_libraries(Starlight.Physics PRIVATE Jolt)
target_compile_definitions(Starlight.Physics PRIVATE NOMINMAX)
